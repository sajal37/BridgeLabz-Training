Q6
a)
ALTER TABLE academics.enrollments 
DROP CONSTRAINT enrollments_course_id_fkey;

ALTER TABLE academics.enrollments 
ADD CONSTRAINT fk_enrollments_course 
FOREIGN KEY (course_id) REFERENCES academics.courses(course_id) ON DELETE RESTRICT;

b)
INSERT INTO academics.enrollments (student_id, course_id, grade)
VALUES ((SELECT student_id FROM academics.students LIMIT 1), 1, 'A');

DELETE FROM academics.courses WHERE course_id = 1;

ERROR: update or delete on table "courses" violates foreign key constraint "fk_enrollments_course" on table "enrollments"
DETAIL: Key (course_id)=(1) is still referenced from table "enrollments".

c)
If we use CASCADE here, deleting a course would automatically delete all student grades and enrollment history for that course. RESTRICT prevents deleting the course by accident, so student academic records are preserved safely.
