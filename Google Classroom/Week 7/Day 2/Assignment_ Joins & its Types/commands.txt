Q1
SELECT s.student_name AS full_name, d.dept_name 
FROM academics.students s
INNER JOIN academics.departments d ON s.dept_id = d.dept_id
ORDER BY d.dept_name, full_name;

Q2

FROM academics.students s
LEFT JOIN academics.departments d ON s.dept_id = d.dept_id;

SELECT s.student_name, d.dept_name 
FROM academics.students s
LEFT JOIN academics.departments d ON s.dept_id = d.dept_id
WHERE s.dept_id IS NULL;

Q3
SELECT 
    s.student_name AS full_name, 
    c.course_name, 
    e.grade, 
    d.dept_name
FROM academics.students s
JOIN academics.enrollments e ON s.student_id = e.student_id
JOIN academics.courses c ON e.course_id = c.course_id
JOIN academics.departments d ON s.dept_id = d.dept_id
WHERE e.grade IS NOT NULL
ORDER BY full_name, c.course_name;

Q6
SELECT d.dept_name, COUNT(s.student_id) AS student_count
FROM academics.departments d
LEFT JOIN academics.students s ON d.dept_id = s.dept_id
GROUP BY d.dept_id, d.dept_name
ORDER BY student_count DESC;

Q7
SELECT c.course_name, s.student_name 
FROM academics.enrollments e
JOIN academics.students s ON e.student_id = s.student_id
RIGHT JOIN academics.courses c ON e.course_id = c.course_id;

Q8
SELECT 
    COALESCE(s.student_id, e.student_id) AS student_id,
    s.student_name AS full_name,
    e.course_id,
    e.grade
FROM academics.students s
FULL OUTER JOIN academics.enrollments e ON s.student_id = e.student_id
WHERE s.student_id IS NULL OR e.student_id IS NULL;

Q9

Q10
SELECT 
    s.student_name AS full_name,
    COUNT(e.course_id) AS courses_enrolled,
    COUNT(e.grade) AS graded_courses
FROM academics.students s
LEFT JOIN academics.enrollments e ON s.student_id = e.student_id
GROUP BY s.student_id, s.student_name;

Q11
a)
SELECT s.student_name AS full_name, d.dept_name 
FROM academics.students s
JOIN academics.departments d USING (dept_id)
ORDER BY d.dept_name, full_name;

b)
USING is convenient when both tables have the exact same column name because it makes the query shorter. It is risky if column names change later or if two tables use the same column name for different meanings.

Q12
SELECT 
    s.student_name,
    sd.dept_name AS student_dept,
    c.course_name,
    cd.dept_name AS course_dept
FROM academics.students s
JOIN academics.departments sd ON s.dept_id = sd.dept_id
JOIN academics.enrollments e ON s.student_id = e.student_id
JOIN academics.courses c ON e.course_id = c.course_id
JOIN academics.departments cd ON c.dept_id = cd.dept_id
WHERE sd.dept_id <> cd.dept_id;
